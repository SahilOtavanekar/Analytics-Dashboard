def campaign_detail_kpis_sql() -> str:
    spans = _session_spans_ctes("ev WHERE SESSION_ID IS NOT NULL")
    return f"""
        WITH ev AS ({_campaign_events_cte()}
        ), sess AS (
            SELECT
                SESSION_ID,
                MAX(IS_AI) AS USED_AI,
                MAX(IS_LEAD) AS CONVERTED,
                MAX(IS_ACTION) AS ACTED,
                MAX(IS_CONSENT_YES) AS SAID_YES,
                MAX(IS_CONSENT_NO) AS SAID_NO,
                MAX(IFF(ASSET IS NOT NULL, 1, 0)) AS SAW_CONTENT,
                MAX(IFF({_ADDR_VALID}, 1, 0)) AS IDENTIFIED
            FROM ev
            WHERE SESSION_ID IS NOT NULL
            GROUP BY SESSION_ID
        ), ev_agg AS (
            -- One pass over the events, not one pass per column. Written as 13 scalar
            -- subqueries first, which measured 4-6s because each re-scanned the CTE;
            -- folded into a single aggregate row it lands near a second. This query is on
            -- the critical path of every drill-in, so it is the one worth flattening.
            SELECT
                MODE(NAME) AS CAMPAIGN_NAME,
                MODE(TENANT) AS TENANT,
                COUNT(DISTINCT MESSAGE_ID) AS EVENTS,
                -- The donut's total and the Interactions / Session numerator - see _INTERACTION_KEY.
                -- EVENTS above stays the raw count: it gates the report and feeds the captions
                -- that are about raw events.
                COUNT(DISTINCT {_INTERACTION_KEY}) AS INTERACTIONS,
                MIN(EVENT_TS)::DATE AS FIRST_SEEN,
                MAX(EVENT_TS)::DATE AS LAST_SEEN,
                COUNT(DISTINCT EVENT_TS::DATE) AS ACTIVE_DAYS,
                COUNT(DISTINCT ASSET) AS ASSETS,
                COUNT(DISTINCT IFF({_ADDR_VALID}, ADDR, NULL)) AS PEOPLE,
                COUNT(DISTINCT IFF({_ADDR_VALID}, SPLIT_PART(ADDR, '@', 2), NULL)) AS COMPANIES,
                COUNT(DISTINCT LEAD_PAGE) AS LEAD_PAGE_SUBMITS,
                -- The read-depth caption's count and the gate on querying that section: the sessions
                -- campaign_read_depth_sql measures, under its own conditions (a page turn, on an
                -- asset, in a session). It was PDF_EVENTS, every page-turn event, which a caption put
                -- at 1,910 over a table built from 1,618 - duplicates and turns with no asset or
                -- session were in the caption and never in the table.
                COUNT(DISTINCT IFF(PAGE_VISITED IS NOT NULL AND ASSET IS NOT NULL, SESSION_ID, NULL)) AS PAGE_TURN_SESSIONS,
                COUNT(DISTINCT IFF(COALESCE(REFERER_URL, '') <> '', REFERER_URL, NULL)) AS REFERRERS,
                -- The gate for the Pages section, counted here rather than asked for separately:
                -- collect() needs to know whether the section has anything in it BEFORE deciding
                -- to query it, and the CTE already projects SEARCH_URL, EVENT_NAME and EVENT_TYPE,
                -- so both expressions resolve against it without a second scan.
                COUNT(DISTINCT IFF({_PAGE_VIEW} AND {_REAL_PAGE}, {_PAGE_URL}, NULL)) AS PAGES,
                -- The Pages caption's total, counted by the interactions rule: once per page per
                -- session (see _INTERACTION_KEY). It was every page-visit event, and 95% of those were
                -- an open tab re-firing - "117,863 page views" sat on the same page as a donut slice
                -- reading "Page visit 5,979". Counted this way the two agree exactly while internal
                -- traffic is excluded. With it included they differ by the page visits on localhost
                -- and hostless pages (285 on that campaign): the donut counts them then, and the Pages
                -- section never does - _REAL_PAGE applies whatever the toggle says.
                COUNT(DISTINCT IFF({_PAGE_VIEW} AND {_REAL_PAGE}, {_INTERACTION_KEY}, NULL)) AS PAGE_VISITS,
                COUNT(DISTINCT TIMEZONE) AS TIMEZONES
            FROM ev
        ), sess_agg AS (
            SELECT
                COUNT(*) AS SESSIONS,
                COALESCE(SUM(USED_AI), 0) AS AI_SESSIONS,
                COALESCE(SUM(CONVERTED), 0) AS LEAD_SESSIONS,
                COALESCE(SUM(ACTED), 0) AS ACTED_SESSIONS,
                COALESCE(SUM(SAID_YES), 0) AS CONSENT_YES_SESSIONS,
                COALESCE(SUM(SAID_NO), 0) AS CONSENT_NO_SESSIONS,
                COALESCE(SUM(SAW_CONTENT), 0) AS CONTENT_SESSIONS,
                COALESCE(SUM(IDENTIFIED), 0) AS IDENTIFIED_SESSIONS
            FROM sess
        ), {spans}, dur_agg AS (
            -- The Session duration captions, measured exactly as the chart's bands are - see
            -- _session_spans_ctes. Interactions per session rather than events for the same reason:
            -- a tab re-firing one page is one interaction, not hundreds of events.
            SELECT
                COALESCE((APPROX_PERCENTILE(SECS, 0.5) / 60.0)::FLOAT, 0) AS MEDIAN_DURATION_MINUTES,
                COALESCE((AVG(SECS) / 60.0)::FLOAT, 0) AS MEAN_DURATION_MINUTES,
                COALESCE(SUM(IFF(INTERACTIONS = 1, 1, 0)), 0) AS SINGLE_INTERACTION_SESSIONS,
                COALESCE((APPROX_PERCENTILE(INTERACTIONS, 0.5))::FLOAT, 0) AS MEDIAN_INTERACTIONS
            FROM spans
        ), ip_pick AS (
            -- Each session's dominant address: the REQUEST_IP on most of its rows, a tie going to
            -- the lower address. Dominant, not MIN, because 2.8% of sessions carry more than one
            -- REQUEST_IP and the one most of the session came from describes it best.
            --
            -- This replaced MODE(REQUEST_IP), which picks the same address but breaks a tie
            -- arbitrarily - and 38 sessions on demand_ai_internal_website_track have two
            -- addresses level, so DISTINCT_IPS read 2,443 on one refresh and 2,444 on the next
            -- from identical data. A figure on the page must not move unless the data does.
            SELECT SESSION_ID, REQUEST_IP AS IP
            FROM ev
            WHERE SESSION_ID IS NOT NULL AND REQUEST_IP IS NOT NULL
            GROUP BY SESSION_ID, REQUEST_IP
            QUALIFY ROW_NUMBER() OVER (PARTITION BY SESSION_ID ORDER BY COUNT(*) DESC, REQUEST_IP) = 1
        ), ip_agg AS (
            -- How concentrated is this campaign's traffic on one network address? Sessions
            -- are the reach figure everything else divides by, and a session count says
            -- nothing about how many PLACES it came from: sas-kr-ai reports 271 sessions,
            -- of which 266 share a single address. Measured across 169 campaigns with 50+
            -- sessions, the top address takes a median 20% but reaches 98% on a handful -
            -- the distribution is bimodal, not a gradient, so a threshold is meaningful here.
            --
            -- An address is NOT a person: a corporate NAT hides a whole office behind one.
            -- That is precisely why this is worth stating rather than charting - it bounds
            -- how the session count may be read, and claims nothing about who.
            SELECT COUNT(*) AS DISTINCT_IPS, COALESCE(MAX(N), 0) AS TOP_IP_SESSIONS
            FROM (SELECT IP, COUNT(*) AS N FROM ip_pick GROUP BY IP)
        )
        SELECT * FROM ev_agg CROSS JOIN sess_agg CROSS JOIN dur_agg CROSS JOIN ip_agg
    """


def campaign_lookup_sql() -> str:
    return f"""
        WITH t AS (
            SELECT MESSAGE_ID, EVENT_TS, PROPERTIES,
                   (EVENT_TS::DATE BETWEEN ? AND ?) AS IN_WINDOW,
                   {campaign_keep_predicate()} AS KEPT
            FROM {TABLE_FQN}
            WHERE CAMPAIGN_ID = ?
              AND EVENT_TS::DATE BETWEEN '2025-06-01' AND CURRENT_DATE()
              AND {_NOT_TEST}
        )
        SELECT
            COUNT(DISTINCT IFF(IN_WINDOW AND KEPT, MESSAGE_ID, NULL)) AS EVENTS_IN_WINDOW,
            COUNT(DISTINCT IFF(IN_WINDOW AND NOT KEPT, MESSAGE_ID, NULL)) AS INTERNAL_EVENTS_IN_WINDOW,
            COUNT(DISTINCT IFF(KEPT, MESSAGE_ID, NULL)) AS EVENTS_EVER,
            COUNT(DISTINCT IFF(NOT KEPT, MESSAGE_ID, NULL)) AS INTERNAL_EVENTS_EVER,
            MIN(IFF(KEPT, EVENT_TS, NULL))::DATE AS FIRST_EVER,
            MAX(IFF(KEPT, EVENT_TS, NULL))::DATE AS LAST_EVER,
            MODE(PROPERTIES:campaign_name::STRING) AS CAMPAIGN_NAME
        FROM t
    """


def campaign_daily_sql() -> str:
    return f"""
        WITH ev AS (
            SELECT
                SESSION_ID, MESSAGE_ID,
                EVENT_TS::DATE AS EVENT_DATE,
                -- The day the session STARTED, carried onto all of its rows.
                MIN(EVENT_TS::DATE) OVER (PARTITION BY SESSION_ID) AS SESSION_START
            FROM {_campaign_table()}
            {_CAMPAIGN_SCOPE.format(not_test=_NOT_TEST)}
        )
        SELECT
            EVENT_DATE,
            -- Each session counted ONCE, on the day it began, so these bars sum to the
            -- Sessions card. A plain COUNT(DISTINCT SESSION_ID) per day counts a session on
            -- every day it touches, so one crossing midnight lands in two bars and the chart
            -- totals more than the card it sits beside. The gap widens with the internal
            -- traffic filter on, because that filter is ROW-level: stripping a session's
            -- staff-tagged rows changes which days it still has rows on.
            COUNT(DISTINCT IFF(EVENT_DATE = SESSION_START, SESSION_ID, NULL)) AS SESSION_COUNT,
            -- Events stay on their own date. An event happened when it happened; only the
            -- session needed attributing to one day.
            COUNT(DISTINCT MESSAGE_ID) AS EVENT_COUNT
        FROM ev
        GROUP BY EVENT_DATE
        ORDER BY EVENT_DATE
    """


def campaign_duration_bands_sql() -> str:
    spans = _session_spans_ctes(f"{_campaign_table()} {_CAMPAIGN_SCOPE.format(not_test=_NOT_TEST)} AND SESSION_ID IS NOT NULL")
    return f"""
        WITH {spans}
        SELECT
            CASE
                WHEN SECS = 0 THEN '0s (instant)'
                WHEN SECS <= 10 THEN '1-10 seconds'
                WHEN SECS <= 60 THEN '10-60 seconds'
                WHEN SECS <= 300 THEN '1-5 minutes'
                WHEN SECS <= 1800 THEN '5-30 minutes'
                WHEN SECS <= 7200 THEN '30 min - 2 hours'
                ELSE 'over 2 hours'
            END AS BAND,
            CASE
                WHEN SECS = 0 THEN 1
                WHEN SECS <= 10 THEN 2
                WHEN SECS <= 60 THEN 3
                WHEN SECS <= 300 THEN 4
                WHEN SECS <= 1800 THEN 5
                WHEN SECS <= 7200 THEN 6
                ELSE 7
            END AS BAND_ORDER,
            COUNT(*) AS SESSIONS
        FROM spans
        GROUP BY BAND, BAND_ORDER
        ORDER BY BAND_ORDER
    """

def _session_spans_ctes(source: str) -> str:
    return f"""firsts AS (
            SELECT SESSION_ID, {_INTERACTION_KEY} AS K, MIN(EVENT_TS) AS T
            FROM {source}
            GROUP BY 1, 2
        ), spans AS (
            SELECT SESSION_ID, DATEDIFF('second', MIN(T), MAX(T)) AS SECS, COUNT(*) AS INTERACTIONS
            FROM firsts
            GROUP BY 1
        )"""

def campaign_geo_sql(timezone_limit: int = 10) -> str:
    return f"""
        WITH base AS (
            SELECT SESSION_ID, TIMEZONE
            FROM {_campaign_table()}
            {_CAMPAIGN_SCOPE.format(not_test=_NOT_TEST)}
              AND SESSION_ID IS NOT NULL
        ), region AS (
            SELECT 'region' AS KIND, {_REGION} AS LABEL, COUNT(DISTINCT SESSION_ID) AS SESSIONS
            FROM base GROUP BY 1, 2
        ), tz AS (
            SELECT 'timezone' AS KIND, TIMEZONE AS LABEL, COUNT(DISTINCT SESSION_ID) AS SESSIONS
            FROM base
            WHERE TIMEZONE IS NOT NULL AND TRIM(TIMEZONE) <> ''
            GROUP BY 1, 2
            QUALIFY ROW_NUMBER() OVER (ORDER BY SESSIONS DESC) <= {timezone_limit}
        )
        SELECT * FROM region
        UNION ALL SELECT * FROM tz
        ORDER BY KIND, SESSIONS DESC
    """


def campaign_pages_sql(limit: int = 12) -> str:
    return f"""
        WITH base AS (
            SELECT SESSION_ID, MESSAGE_ID, EVENT_NAME, EVENT_TYPE, SEARCH_URL,
                   {_PAGE_SCHEME} AS SCHEME
            FROM {_campaign_table()}
            {_CAMPAIGN_SCOPE.format(not_test=_NOT_TEST)}
              AND SESSION_ID IS NOT NULL
        ), all_sessions AS (
            SELECT COUNT(DISTINCT SESSION_ID) AS N FROM base
        ), grouped AS (
            SELECT
                {_PAGE_HOST} AS HOST,
                '/' || {_PAGE_PATH} AS PATH,
                {_PAGE_URL} AS PAGE_URL,
                MODE(SCHEME) AS SCHEME,
                COUNT(DISTINCT SESSION_ID) AS SESSIONS,
                COALESCE((COUNT(DISTINCT SESSION_ID) * 100.0
                    / NULLIF((SELECT N FROM all_sessions), 0))::FLOAT, 0) AS PCT_OF_SESSIONS
            FROM base
            WHERE {_PAGE_VIEW}
              AND SEARCH_URL IS NOT NULL
              AND {_REAL_PAGE}
            GROUP BY 1, 2, 3
        )
        SELECT
            IFF(COUNT(*) OVER (PARTITION BY PATH) > 1, PAGE_URL, PATH) AS PAGE,
            SESSIONS,
            PCT_OF_SESSIONS,
            HOST,
            SCHEME || '://' || HOST || PATH AS URL
        FROM grouped
        ORDER BY SESSIONS DESC
        LIMIT {limit}
    """


def campaign_assets_sql(limit: int = 12) -> str:
    return f"""
        SELECT
            {_ASSET_LABEL} AS ASSET,
            COUNT(DISTINCT SESSION_ID) AS SESSIONS,
            COUNT(DISTINCT {_INTERACTION_KEY}) AS INTERACTIONS,
            ROUND(COUNT(DISTINCT {_INTERACTION_KEY}) * 1.0
                  / NULLIF(COUNT(DISTINCT SESSION_ID), 0), 1) AS INTERACTIONS_PER_SESSION
        FROM {_campaign_table()}
        {_CAMPAIGN_SCOPE.format(not_test=_NOT_TEST)}
          AND {_ASSET} IS NOT NULL
          AND SESSION_ID IS NOT NULL
        GROUP BY {_ASSET}
        ORDER BY SESSIONS DESC
        LIMIT {limit}
    """


def campaign_read_depth_sql(limit: int = 10) -> str:
    return f"""
        WITH per_session AS (
            SELECT
                {_ASSET} AS ASSET_KEY,
                {_ASSET_LABEL} AS ASSET,
                SESSION_ID,
                MAX(TRY_TO_NUMBER(PROPERTIES:page_visited::STRING)) AS DEEPEST
            FROM {_campaign_table()}
            {_CAMPAIGN_SCOPE.format(not_test=_NOT_TEST)}
              AND PROPERTIES:page_visited IS NOT NULL
              AND {_ASSET} IS NOT NULL
              AND SESSION_ID IS NOT NULL
            GROUP BY {_ASSET}, SESSION_ID
        )
        SELECT
            ASSET,
            COUNT(*) AS SESSIONS,
            ROUND(AVG(DEEPEST), 1) AS AVG_PAGE_REACHED,
            MAX(DEEPEST) AS DEEPEST_PAGE
        FROM per_session
        GROUP BY ASSET_KEY, ASSET
        ORDER BY SESSIONS DESC
        LIMIT {limit}
    """


def campaign_companies_sql(limit: int = 12) -> str:
    return f"""
        SELECT
            {_DOMAIN} AS COMPANY,
            COUNT(DISTINCT {_EMAIL}) AS PEOPLE,
            COUNT(DISTINCT SESSION_ID) AS SESSIONS
        FROM {_campaign_table()}
        {_CAMPAIGN_SCOPE.format(not_test=_NOT_TEST)}
          AND {_VALID_EMAIL}
          AND {_DOMAIN} NOT IN {_FREE_MAIL}
          AND SESSION_ID IS NOT NULL
        GROUP BY 1
        ORDER BY SESSIONS DESC
        LIMIT {limit}
    """


_ARRIVAL_GROUP = f"""
        CASE
            WHEN COALESCE(REFERER_URL, '') = '' THEN 'Direct'
            WHEN ({_REF_HOST} = {_PAGE_HOST} AND {_PAGE_HOST} <> '') OR {_INTERNAL_REFERRER} THEN 'Internal'
            ELSE 'External'
        END"""


def campaign_sources_sql(referrer_limit: int = 8) -> str:
    return f"""
        WITH arrivals AS (
            SELECT REFERER_URL, SEARCH_URL
            FROM {_campaign_table()}
            {_CAMPAIGN_SCOPE.format(not_test=_NOT_TEST)}
              AND SESSION_ID IS NOT NULL
            QUALIFY ROW_NUMBER() OVER (PARTITION BY SESSION_ID ORDER BY EVENT_TS, MESSAGE_ID) = 1
        ), grp AS (
            SELECT 'group' AS KIND, {_ARRIVAL_GROUP} AS LABEL, COUNT(*) AS SESSIONS
            FROM arrivals
            WHERE {_ARRIVAL_GROUP} <> 'Internal'
            GROUP BY 1, 2
        ), ref AS (
            SELECT 'referrer' AS KIND, {_REF_HOST} AS LABEL, COUNT(*) AS SESSIONS
            FROM arrivals
            WHERE {_ARRIVAL_GROUP} = 'External' AND {_REF_HOST} IS NOT NULL
            GROUP BY 1, 2
            QUALIFY ROW_NUMBER() OVER (ORDER BY SESSIONS DESC, LABEL) <= {referrer_limit}
        )
        SELECT * FROM grp
        UNION ALL SELECT * FROM ref
        -- LABEL breaks ties so two referrers with equal sessions keep one order. Without it the
        -- order among them followed the query plan, and moved when this SQL's text did.
        ORDER BY KIND, SESSIONS DESC, LABEL
    """


def campaign_ai_sql() -> str:
    return f"""
        SELECT
            {_MODEL} AS MODEL,
            COUNT(DISTINCT SESSION_ID) AS SESSIONS,
            COUNT(DISTINCT {_INTERACTION_KEY}) AS INTERACTIONS
        FROM {_campaign_table()}
        {_CAMPAIGN_SCOPE.format(not_test=_NOT_TEST)}
          AND {_AI_REQUEST}
          AND SESSION_ID IS NOT NULL
        GROUP BY 1
        ORDER BY SESSIONS DESC, MODEL
    """

def _as_date(value):
    if hasattr(value, "year") and hasattr(value, "month"):
        return value
    try:
        return dt.date.fromisoformat(str(value)[:10])
    except ValueError:
        return None


def _fmt_date(value, year: bool = False) -> str:
    d = _as_date(value)
    if d is None:
        return str(value)
    return f"{d.day} {_MONTH[d.month - 1]}" + (f" {d.year}" if year else "")


def _fmt_range(start, end) -> str:
    a, b = _as_date(start), _as_date(end)
    if a is None or b is None:
        return f"{start} to {end}"
    if a.year != b.year:
        return f"{_fmt_date(a, year=True)} - {_fmt_date(b, year=True)}"
    return f"{_fmt_date(a)} - {_fmt_date(b)} {b.year}"


def classify_lookup(found) -> str:
    """open / internal_only (everything left is internal traffic the filter removed) /
    wrong_window (a real campaign on other dates) / unknown (no such id)."""
    if int(found["EVENTS_IN_WINDOW"]) > 0:
        return "open"
    if int(found.get("INTERNAL_EVENTS_IN_WINDOW", 0) or 0) > 0:
        return "internal_only"
    if int(found["EVENTS_EVER"]) > 0:
        return "wrong_window"
    if int(found.get("INTERNAL_EVENTS_EVER", 0) or 0) > 0:
        return "internal_only"
    return "unknown"


def internal_only_message(found, window: str) -> str:
    n = int(found.get("INTERNAL_EVENTS_IN_WINDOW", 0) or 0)
    return _INTERNAL_IN_WINDOW_MSG.format(n=n, window=window) if n else _INTERNAL_EVER_MSG.format(window=window)


def partial_day(daily, today):
    """The activity chart's last day if it has not finished yet (UTC), else None."""
    if daily is None or not len(daily):
        return None
    days = [d for d in (_as_date(v) for v in daily["EVENT_DATE"]) if d is not None]
    last = max(days) if days else None
    return last if last is not None and last >= today else None


def partial_day_note(day) -> str:
    return PARTIAL_DAY_NOTE.format(day=_fmt_date(day))


def concentration(kpis):
    sessions = int(kpis["SESSIONS"] or 0)
    top = int(kpis["TOP_IP_SESSIONS"] or 0)
    if sessions < CONCENTRATION_MIN_SESSIONS or not sessions:
        return None
    share = top / sessions * 100
    return share if share >= CONCENTRATION_PCT else None


def pct(part, whole) -> float:
    part, whole = float(part or 0), float(whole or 0)
    return part / whole * 100 if whole else 0.0


def collect(run, campaign_id: str, start: dt.date, end: dt.date) -> dict:
    """Every section of one campaign's report, gated - identical to src/campaign_report.py.
    `run` is (sql, params) -> DataFrame."""
    params = [start, end, campaign_id]
    kpis = run(campaign_detail_kpis_sql(), params).iloc[0]
    out: dict = {
        "campaign_id": campaign_id,
        "start": start,
        "end": end,
        "kpis": kpis,
        "label": kpis["CAMPAIGN_NAME"] or campaign_id,
        "named": bool(kpis["CAMPAIGN_NAME"]),
    }

    if int(kpis["EVENTS"]) == 0:
        found = run(campaign_lookup_sql(), params).iloc[0]
        out["status"] = STATUS_MISSING
        out["reason"] = classify_lookup(found)
        out["lookup"] = found
        return out

    if int(kpis["SESSIONS"]) == 0:
        out["status"] = STATUS_NO_SESSIONS
        return out

    out["status"] = STATUS_OK
    out["daily"] = run(campaign_daily_sql(), params)
    out["partial_day"] = partial_day(out["daily"], dt.datetime.now(dt.timezone.utc).date())
    out["duration"] = run(campaign_duration_bands_sql(), params)
    out["event_mix"] = run(campaign_event_mix_sql(), params)

    geo = run(campaign_geo_sql(), params)
    out["regions"] = geo[geo["KIND"] == "region"][["LABEL", "SESSIONS"]].rename(columns={"LABEL": "REGION"})
    out["timezones"] = geo[geo["KIND"] == "timezone"][["LABEL", "SESSIONS"]].rename(columns={"LABEL": "TIMEZONE"})

    if int(kpis["PAGES"]) > 0:
        out["pages"] = run(campaign_pages_sql(), params)

    if int(kpis["ASSETS"]) > 0:
        out["assets"] = run(campaign_assets_sql(), params)
        if int(kpis["PAGE_TURN_SESSIONS"]) > 0:
            out["depth"] = run(campaign_read_depth_sql(), params)

    if int(kpis["PEOPLE"]) >= IDENTITY_FLOOR:
        out["companies"] = run(campaign_companies_sql(), params)

    sources = run(campaign_sources_sql(), params)
    out["sources"] = sources[sources["KIND"] == "group"][["LABEL", "SESSIONS"]].rename(columns={"LABEL": "SOURCE_GROUP"})
    out["referrers"] = sources[sources["KIND"] == "referrer"][["LABEL", "SESSIONS"]].rename(columns={"LABEL": "REFERRER"})

    if int(kpis["AI_SESSIONS"]) > 0:
        out["ai"] = run(campaign_ai_sql(), params)

    return out

def _blank(v) -> bool:
    """None, or pandas' NaN for a missing value - NaN is truthy and prints as "nan"."""
    return v is None or (isinstance(v, float) and v != v)


def _esc(v) -> str:
    return html.escape("" if _blank(v) else str(v), quote=True)


def _html_address(v) -> str:
    """A path or host, escaped, with a <wbr> break after each separator. A browser breaks a
    long address only after a hyphen, so without these a long host runs off the right edge.
    Split before escaping so a break never lands inside an entity such as &amp;."""
    return "<wbr>".join(_esc(part) for part in re.split("(?<=[/.?&=-])", "" if _blank(v) else str(v)))

def _bar_rows(frame, label_col, value_col, extra=None, link_col=None, wrap=False) -> str:
    if frame is None or not len(frame):
        return ""
    top = float(max(float(v or 0) for v in frame[value_col])) or 1.0
    out = []
    extra_class = "n w" if wrap else "n"
    for row in frame.itertuples():
        label = _html_address(getattr(row, label_col)) if wrap else _esc(getattr(row, label_col))
        if link_col:
            href = _esc(getattr(row, link_col, ""))
            if href:
                label = f"<a href='{href}'>{label}</a>"
        value = float(getattr(row, value_col) or 0)
        width = max(value / top * 100.0, 0.0)
        cells = f"<td>{label}</td><td class='n'>{value:,.10g}</td>"
        cells += f"<td class='barcell'><div class='bar' style='width:{width:.1f}%'></div></td>"
        if extra:
            for col, fmt in extra:
                cells += f"<td class='{extra_class}'>{fmt(getattr(row, col))}</td>"
        out.append(f"<tr>{cells}</tr>")
    return "".join(out)

def _table(title, frame, label_col, value_col, value_head, extra=None, note=None, link_col=None, wrap=False) -> str:
    if frame is None or not len(frame):

def _png(width: int, height: int, rows) -> bytes:
    """An RGBA PNG from rows of bytes, written with zlib alone - no imaging package to install."""
    def chunk(tag, data):
        return struct.pack(">I", len(data)) + tag + data + struct.pack(">I", zlib.crc32(tag + data) & 0xFFFFFFFF)
    raw = b"".join(bytes([0]) + bytes(row) for row in rows)
    header = struct.pack(">IIBBBBB", width, height, 8, 6, 0, 0, 0)
    return bytes([137, 80, 78, 71, 13, 10, 26, 10]) + chunk(b"IHDR", header) + chunk(b"IDAT", zlib.compress(raw, 9)) + chunk(b"IEND", b"")


def _ring_png(slices, size=320, r_out=150.0, r_in=94.0, gap=2.5) -> bytes:
    """The interactions ring as a PNG: slices clockwise from 12 o'clock, as on the dashboard and
    the PDF. Drawn at twice its 160px display size and anti-aliased with 3x3 sub-samples per pixel;
    the background is transparent. `slices` is [(fraction, "#rrggbb")]."""
    bounds, acc = [], 0.0
    for frac, hue in slices:
        rgb = tuple(int(hue[i:i + 2], 16) for i in (1, 3, 5))
        bounds.append((acc, acc + frac, rgb))
        acc += frac
    edges = [a for a, _, _ in bounds] if len(bounds) > 1 else []
    c = size / 2.0
    offsets = (-1 / 3, 0.0, 1 / 3)
    rows = []
    for y in range(size):
        row = bytearray(size * 4)
        for x in range(size):
            dx0, dy0 = x + 0.5 - c, y + 0.5 - c
            d = math.hypot(dx0, dy0)
            if d < r_in - 1 or d > r_out + 1:
                continue
            red = green = blue = hits = 0
            for oy in offsets:
                for ox in offsets:
                    dx, dy = dx0 + ox, dy0 + oy
                    r = math.hypot(dx, dy)
                    if r < r_in or r > r_out:
                        continue
                    t = (math.atan2(dx, -dy) / (2 * math.pi)) % 1.0
                    if any(min(abs(t - e), 1 - abs(t - e)) * 2 * math.pi * r < gap / 2 for e in edges):
                        continue
                    rgb = next((col for a, b, col in bounds if a <= t < b), bounds[-1][2])
                    red, green, blue, hits = red + rgb[0], green + rgb[1], blue + rgb[2], hits + 1
            if hits:
                i = x * 4
                row[i:i + 4] = bytes((red // hits, green // hits, blue // hits, round(255 * hits / 9)))
        rows.append(row)
    return _png(size, size, rows)


_INK = "#0b1a1a"
_MUTED = "#546a6a"


def _readable_on(hue: str) -> str:
    """Ink or white, whichever has the higher contrast against `hue` (WCAG relative luminance) -
    the dashboard's rule for text on a slice: dark on the teal, white on the purple."""
    def lum(h):
        chans = [int(h[i:i + 2], 16) / 255 for i in (1, 3, 5)]
        lin = [c / 12.92 if c <= 0.03928 else ((c + 0.055) / 1.055) ** 2.4 for c in chans]
        return 0.2126 * lin[0] + 0.7152 * lin[1] + 0.0722 * lin[2]
    bg = lum(hue)
    on_white = 1.05 / (bg + 0.05)
    on_ink = (bg + 0.05) / (lum(_INK) + 0.05)
    return "#ffffff" if on_white >= on_ink else _INK


def _ring_png_styled(rows, total_txt) -> bytes:
    """The PDF's ring as a PNG: slices clockwise from 12 o'clock with a thin white gap between them,
    and the total over "INTERACTIONS" in the hole. Shares are in the legend beside it, as in the PDF.
    Drawn with matplotlib's non-interactive Agg backend at 400px, shown at 200px so it stays sharp."""

def _html_mix(mix, total_interactions, images=None) -> str:
    """The interactions split - Page visit / Clicks / Form submit - as the dashboard's ring.

    The ring used to be inline SVG, which mail clients strip: a real test email showed only its two
    text labels, run together as "7,928INTERACTIONS". It is now a PNG drawn here and attached to the
    email inline (cid:), which Outlook and Gmail both display; the total stands beside it as text
    and the legend carries every count and share. `images` collects the PNG for run_report() to
    attach. If drawing fails, or no `images` dict is passed, the split falls back to a stacked bar
    built from table cells, so the email still goes out with the numbers.
    """
    if mix is None or not len(mix) or float(mix["INTERACTIONS"].sum()) <= 0:
        return ""
    by_bucket = {str(r.BUCKET): float(r.INTERACTIONS or 0) for r in mix.itertuples()}
    rows = [(name, by_bucket.get(name, 0.0), _MIX_HUES[i % len(_MIX_HUES)]) for i, name in enumerate(EVENT_BUCKET_ORDER) if by_bucket.get(name, 0.0) > 0]
    if not rows:
        return ""
    shown = sum(v for _, v, _ in rows)
    legend, segments = [], []
    for label, value, hue in rows:
        share = value / shown * 100.0
        share_txt = f"{share:.1f}%" if share >= 0.1 else "&lt;0.1%"
        legend.append(_LEG_ROW.format(hue=hue, label=_esc(label), count=f"{int(round(value)):,}", share=share_txt))
        # A sliver still gets half a percent so it stays visible; the legend has the exact share.
        segments.append(_MIX_SEGMENT.format(w=f"{max(share, 0.5):.1f}", hue=hue))
    total_txt = f"{int(round(shown)):,}"
    missing = int(total_interactions) - int(round(shown))
    gap = f"<p class='note'>{total_txt} of this campaign's {int(total_interactions):,} interactions fall into these three buckets; {missing:,} do not.</p>" if missing > 0 else ""
    legend_cell = _LEG_CELL.format(rows="".join(legend))

    if images is not None:
        try:
            styled = _ring_png_styled(rows, total_txt)
        except Exception:
            styled = None
        if styled is not None:
            images[_RING_CID] = styled
            alt = _esc(", ".join(f"{n} {int(round(v)):,}" for n, v, _ in rows))
            return ("<table class='mix' cellpadding='0' cellspacing='0' border='0'><tr>"
                    + _MIX_RING_STYLED.format(cid=_RING_CID, alt=alt)
                    + legend_cell + f"</tr></table>{gap}")

    # Fallbacks: a plain ring with the total beside it, then the stacked bar.
    ring = None
    if images is not None:
        try:
            # A sliver keeps at least 1% of the circle so it stays visible; the legend has the exact share.
            fracs = [max(v / shown, 0.01) for _, v, _ in rows]
            scale = sum(fracs)
            ring = _ring_png([(f / scale, hue) for f, (_, _, hue) in zip(fracs, rows)])
        except Exception:
            ring = None
    if ring is not None:
        images[_RING_CID] = ring
        alt = _esc(", ".join(f"{n} {int(round(v)):,}" for n, v, _ in rows))
        return ("<table class='mix' cellpadding='0' cellspacing='0' border='0'><tr>"
                + _MIX_RING.format(cid=_RING_CID, alt=alt)
                + _MIX_TOTAL.format(total=total_txt)
                + legend_cell + f"</tr></table>{gap}")

    bar = ("<table cellpadding='0' cellspacing='0' border='0' style='width:100%;border-collapse:collapse;table-layout:fixed'>"
           f"<tr>{''.join(segments)}</tr></table>")
    return ("<table class='mix' cellpadding='0' cellspacing='0' border='0'><tr>"
            + _MIX_TOTAL.format(total=total_txt)
            + legend_cell + "</tr>"
            + f"<tr><td colspan='2' style='padding:0 16px 16px'>{bar}</td></tr></table>{gap}")


def _short_date(v) -> str:
    d = _as_date(v)
    return _esc(v) if d is None else f"{d.day} {_MONTH[d.month - 1]}"


_DAY_VALUE = "<td style='text-align:center;font-size:9px;{weight}color:#0b1a1a;padding:0 1px;white-space:nowrap;'>{text}</td>"
_DAY_EMPTY = ("<td style='padding:0 1px;vertical-align:bottom;'><table cellpadding='0' cellspacing='0' border='0' "
              "style='width:100%;border-collapse:collapse'><tr><td height='{h}' style='height:{h}px;line-height:1px;"
              "font-size:1px;'>&nbsp;</td></tr></table></td>")
_DAY_BAR = ("<td style='padding:0 1px;vertical-align:bottom;'><table cellpadding='0' cellspacing='0' border='0' "
            "style='width:100%;border-collapse:collapse'><tr><td height='{spacer}' style='height:{spacer}px;"
            "line-height:1px;font-size:1px;'>&nbsp;</td></tr><tr><td height='{bar}' bgcolor='{hue}' "
            "style='height:{bar}px;line-height:1px;font-size:1px;background-color:{hue};'>&nbsp;</td></tr></table></td>")
_DAY_LABEL = "<td style='text-align:center;font-size:8px;color:#546a6a;padding:2px 1px 0;white-space:nowrap;'>{text}</td>"


def _daily_chart(frame, label_col, value_col, partial=None) -> str:
    """Activity over time as an Outlook-safe column chart: nested tables with the height=
    and bgcolor= attributes, the two things Outlook's Word engine actually renders.

    Matches the PDF's chart rules: every calendar day from the first to the last keeps its
    slot, so a quiet gap shows as a gap rather than closing up; today's bar, still filling
    (UTC), is drawn lighter and noted; Quietest is chosen from finished days only.
    """
    if frame is None or not len(frame):
        return ""
    by_day = {}
    for row in frame.itertuples():
        d = _as_date(getattr(row, label_col))
        if d is not None:
            by_day[d] = float(getattr(row, value_col) or 0)
    if not by_day:
        return ""
    first, last = min(by_day), max(by_day)
    days = [first + dt.timedelta(days=i) for i in range((last - first).days + 1)]
    active = [d for d in days if d in by_day]
    top = max(by_day.values()) or 1.0
    peak = max(active, key=lambda d: by_day[d])
    finished = [d for d in active if partial is None or d < partial]
    quiet = min(finished, key=lambda d: by_day[d]) if finished else None
    max_px = 110
    label_every = max(1, round(len(days) / 15))

    val_cells, bar_cells, date_cells = [], [], []
    for i, d in enumerate(days):
        v = by_day.get(d)
        weight = "font-weight:700;" if d in (peak, quiet) else ""
        val_cells.append(_DAY_VALUE.format(weight=weight, text="" if v is None else f"{v:,.0f}"))
        if v is None:
            bar_cells.append(_DAY_EMPTY.format(h=max_px))
        else:
            bar_px = max(round(v / top * max_px), 1)
            hue = _TEAL_PARTIAL if d == partial else _TEAL
            bar_cells.append(_DAY_BAR.format(spacer=max_px - bar_px, bar=bar_px, hue=hue))
        show = (i % label_every == 0) or i == len(days) - 1
        date_cells.append(_DAY_LABEL.format(text=_short_date(d) if show else ""))

    chart = ("<table cellpadding='0' cellspacing='0' border='0' style='width:100%;table-layout:fixed;border-collapse:collapse'>"
             f"<tr>{''.join(val_cells)}</tr><tr>{''.join(bar_cells)}</tr><tr>{''.join(date_cells)}</tr></table>")
    total = sum(by_day.values())
    days_txt = f"{len(days):,} days" if len(active) == len(days) else f"{len(active):,} active days of {len(days):,}"
    quiet_txt = f" &middot; Quietest {by_day[quiet]:,.0f} on {_short_date(quiet)}" if quiet is not None else ""
    caption = f"<p class='note'>Sessions: {total:,.0f} over {days_txt} &middot; Peak {by_day[peak]:,.0f} on {_short_date(peak)}{quiet_txt}</p>"
    note = f"<p class='note'>{_esc(partial_day_note(partial))}</p>" if partial is not None and partial in by_day else ""
    return f"<h2>Activity over time</h2>{note}{chart}{caption}"


def _ident(data) -> str:
    """The identity line under the title, as the PDF writes it."""
    k = data["kpis"]
    facts = [f"Campaign ID <code>{_esc(data['campaign_id'])}</code>"]
    if not _blank(k["TENANT"]) and k["TENANT"]:
        facts.append(f"Tenant <code>{_esc(k['TENANT'])}</code>")
    facts.append(f"Reporting on {_esc(_fmt_range(data['start'], data['end']))}")
    window_days = (_as_date(data["end"]) - _as_date(data["start"])).days + 1
    active = f"active on {int(k['ACTIVE_DAYS']):,} of {window_days:,} days"
    if _as_date(k["FIRST_SEEN"]) != _as_date(data["start"]) or _as_date(k["LAST_SEEN"]) != _as_date(data["end"]):
        active = f"ran {_esc(_fmt_range(k['FIRST_SEEN'], k['LAST_SEEN']))}, {active}"
    facts.append(active[0].upper() + active[1:])
    return " &middot; ".join(facts)

        f"<title>{_esc(title)} - campaign report</title><style>{_CSS}</style></head>"
        f"<body><div class='wrap'>{body}</div></body></html>"
    )


def to_html(data: dict, link: str = "", images=None) -> str:

    label, cid = _esc(data["label"]), _esc(data["campaign_id"])
    window = f"{data['start']} to {data['end']}"

    if data["status"] == STATUS_MISSING:
        if data.get("reason") == "internal_only":
            msg = _esc(internal_only_message(data["lookup"], f"{data['start']} and {data['end']}"))
        elif data.get("reason") == "wrong_window":
            f = data["lookup"]
            msg = (f"Campaign <code>{cid}</code> has no activity between {window}. "
                   f"It ran <b>{_esc(f['FIRST_EVER'])} to {_esc(f['LAST_EVER'])}</b> "
                   f"with {int(f['EVENTS_EVER']):,} events.")
        else:
            msg = f"No campaign <code>{cid}</code> exists in the tracked data."
        return _document(label, f"<h1>{label}</h1><div class='warn'>{msg}</div>", window, link)

    if data["status"] == STATUS_NO_SESSIONS:
        body = (f"<h1>{label}</h1><p class='sub'>Campaign ID {cid}</p><div class='warn'>"
                f"{int(k['EVENTS']):,} events recorded between {window}, but no identifiable "
                f"sessions: session IDs were not captured on campaign rows before November 2025, "
                f"so only event counts exist this far back.</div>")
        return _document(label, body, window, link)

    s = int(k["SESSIONS"])
    parts = [f"<h1>{label}</h1>", f"<p class='sub'>{_ident(data)}</p>"]

    # The PDF's top row: the two cards stacked on the left, the ring and its legend in a box on the right.
    cards = _CARD.format(label="Distinct sessions", value=f"{s:,}") + _CARD_GAP + _CARD.format(label="Interactions / session", value=f"{float(k['INTERACTIONS']) / s:,.1f}")

                        extra=[("PEOPLE", _fmt_count)], note=COMPANIES_NOTE))
    people = int(k["PEOPLE"])
    if data.get("companies") is None and 0 < people < IDENTITY_FLOOR:
        noun = "person" if people == 1 else "people"
        parts.append(_section_note("Accounts reached", f"Company breakdown withheld: only {people} identified {noun} in this range, too few to name a company without describing an individual."))

    parts.append(_table("How they arrived", data.get("sources"), "SOURCE_GROUP", "SESSIONS", "Sessions", note=ARRIVAL_NOTE))

    parts.append(_table("AI model mix", data.get("ai"), "MODEL", "SESSIONS", "Sessions",

# 5. The reporting window - verbatim from src/components.py's date_range_filter():
#    the window a fresh dashboard session opens with, so the email and the
#    dashboard opened the same day show the same dates.

_DATA_STARTS = dt.date(2025, 6, 1)
_PRESET_MONTHS = (1, 3, 6, 12, 24)


def _months_before(day: dt.date, months: int) -> dt.date:
    total = day.year * 12 + day.month - 1 - months
    year, month = divmod(total, 12)
    return dt.date(year, month + 1, min(day.day, calendar.monthrange(year, month + 1)[1]))


def _preset_range(today: dt.date, months: int) -> tuple:
    return (max(_months_before(today, months) + dt.timedelta(days=1), _DATA_STARTS), today)


def _dashboard_window(today: dt.date, default_days: int) -> tuple:
    """"30 days" snaps to the nearest whole-month preset when that preset starts within two
    days of the target - which is why the window is 28 to 31 days long depending on the month."""
    target = (today - dt.timedelta(days=default_days), today)
    gap, months = min((abs((_preset_range(today, m)[0] - target[0]).days), m) for m in _PRESET_MONTHS)
    chosen = _preset_range(today, months) if gap <= 2 else target
    latest = today + dt.timedelta(days=1)
    return (min(max(chosen[0], _DATA_STARTS), latest), min(max(chosen[1], _DATA_STARTS), latest))

    msg = MIMEMultipart("related")
    msg["Subject"] = subject
    msg["From"] = sender
    msg["To"] = ", ".join(recipients)
    if ccrecipients:
        msg["Cc"] = ", ".join(ccrecipients)
    msg.attach(MIMEText(email_html, "html", "utf-8"))
    for cid, png in images.items():
        part = MIMEImage(png, "png")

    ses = boto3.client("sesv2", region_name=region, aws_access_key_id=access_key, aws_secret_access_key=secret_key)
    ses.send_email(
        FromEmailAddress=sender,
        Destination={"ToAddresses": recipients, "CcAddresses": ccrecipients},
        Content={"Raw": {"Data": msg.as_bytes()}},
    )

    return (f"SES email sent to: {', '.join(recipients)} | campaign={campaign_id} "
            f"status={data['status']} exclude_internal={exclude_internal_flag} "
            f"window={start_date} to {end_date}")
